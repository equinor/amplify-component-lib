## Usage

Popovers are used to display non-critical, extra information or actions floating near a target element.

```tsx
import { Popover, Button, Typography } from '@equinor/amplify-component-lib';
import { useRef, useState } from 'react';

const MyComponent = () => {
  const [isOpen, setIsOpen] = useState(false);
  const anchorRef = useRef<HTMLButtonElement>(null);

  return (
    <>
      <Button ref={anchorRef} onClick={() => setIsOpen(!isOpen)}>
        Toggle Popover
      </Button>
      <Popover
        open={isOpen}
        anchorEl={anchorRef.current}
        onClose={() => setIsOpen(false)}
      >
        <Popover.Header>
          <Popover.Title>Popover Title</Popover.Title>
        </Popover.Header>
        <Popover.Content>
          <Typography variant="body_short">
            Popover content goes here.
          </Typography>
        </Popover.Content>
        <Popover.Actions>
          <Button onClick={() => setIsOpen(false)}>Close</Button>
        </Popover.Actions>
      </Popover>
    </>
  );
};
```
